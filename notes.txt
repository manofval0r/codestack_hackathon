class GameState {
  constructor() {
    this.state = this.loadState() || this.getInitialState();
    this.listeners = new Map();
  }
  
  getInitialState() {
    return {
      player: {
        id: this.generatePlayerId(),
        name: '',
        level: 1,
        xp: 0,
        totalXP: 0,
        currentChallenge: null,
        selectedPath: null, // 'fundamentals' or 'ai-collaboration'
        sessionStartTime: Date.now(),
        lastActiveTime: Date.now()
      },
      
      progression: {
        completedChallenges: new Set(),
        unlockedLevels: new Set([1]),
        currentStreak: 0,
        longestStreak: 0,
        achievements: new Set()
      },
      
      choices: {
        aiAssistanceUsed: 0,
        manualSolvesCompleted: 0,
        hintsRequested: 0,
        tabSwitchDetected: 0,
        reflectionPrompts: []
      },
      
      performance: {
        averageCompletionTime: 0,
        accuracyRate: 0,
        improvementRate: 0,
        aiDependencyScore: 0
      },
      
      preferences: {
        theme: 'dark',
        soundEnabled: true,
        hintPreference: 'gradual',
        privacyMode: false
      }
    };
  }
  
  saveState() {
    try {
      const serializedState = {
        ...this.state,
        progression: {
          ...this.state.progression,
          completedChallenges: Array.from(this.state.progression.completedChallenges),
          unlockedLevels: Array.from(this.state.progression.unlockedLevels),
          achievements: Array.from(this.state.progression.achievements)
        }
      };
      localStorage.setItem('debug-quest-save', JSON.stringify(serializedState));
    } catch (error) {
      console.warn('Failed to save game state:', error);
    }
  }
  
  loadState() {
    try {
      const saved = localStorage.getItem('debug-quest-save');
      if (saved) {
        const parsedState = JSON.parse(saved);
        // Convert arrays back to Sets
        parsedState.progression.completedChallenges = new Set(parsedState.progression.completedChallenges);
        parsedState.progression.unlockedLevels = new Set(parsedState.progression.unlockedLevels);
        parsedState.progression.achievements = new Set(parsedState.progression.achievements);
        return parsedState;
      }
    } catch (error) {
      console.warn('Failed to load game state:', error);
    }
    return null;
  }
  
  updateState(path, value) {
    const keys = path.split('.');
    let current = this.state;
    
    for (let i = 0; i < keys.length - 1; i++) {
      if (!current[keys[i]]) current[keys[i]] = {};
      current = current[keys[i]];
    }
    
    current[keys[keys.length - 1]] = value;
    this.saveState();
    this.notifyListeners(path, value);
  }
  
  subscribe(path, callback) {
    if (!this.listeners.has(path)) {
      this.listeners.set(path, new Set());
    }
    this.listeners.get(path).add(callback);
  }
  
  notifyListeners(path, value) {
    const callbacks = this.listeners.get(path);
    if (callbacks) {
      callbacks.forEach(callback => callback(value, path));
    }
  }
}

class ChallengeManager {
  constructor() {
    this.challenges = new Map();
    this.currentChallenge = null;
    this.loadChallenges();
  }
  
  async loadChallenges() {
    try {
      const challengeFiles = [
        'level-01-html.json',
        'level-02-css.json', 
        'level-03-js.json',
        'level-04-python.json',
        'level-05-react.json'
      ];
      
      const challengeData = await Promise.all(
        challengeFiles.map(file => 
          fetch(`./data/challenges/${file}`).then(r => r.json())
        )
      );
      
      challengeData.forEach(levelData => {
        levelData.challenges.forEach(challenge => {
          this.challenges.set(challenge.id, {
            ...challenge,
            level: levelData.level,
            prerequisites: levelData.prerequisites || [],
            category: levelData.category
          });
        });
      });
      
    } catch (error) {
      console.error('Failed to load challenges:', error);
      this.loadFallbackChallenges();
    }
  }
  
  getChallengesByLevel(level) {
    return Array.from(this.challenges.values())
      .filter(challenge => challenge.level === level);
  }
  
  getChallenge(challengeId) {
    return this.challenges.get(challengeId);
  }
  
  validateSolution(challengeId, solution) {
    const challenge = this.getChallenge(challengeId);
    if (!challenge) return { valid: false, error: 'Challenge not found' };
    
    // Solution validation logic based on challenge type
    switch (challenge.type) {
      case 'bug-fix':
        return this.validateBugFix(challenge, solution);
      case 'code-completion':
        return this.validateCodeCompletion(challenge, solution);
      case 'prompt-crafting':
        return this.validatePromptCrafting(challenge, solution);
      default:
        return { valid: false, error: 'Unknown challenge type' };
    }
  }
  
  validateBugFix(challenge, solution) {
    // Multiple validation approaches for robustness
    const validations = {
      exactMatch: challenge.solutions.exact?.includes(solution.trim()),
      regexMatch: challenge.solutions.patterns?.some(pattern => 
        new RegExp(pattern, 'i').test(solution)
      ),
      functionalTest: this.runFunctionalTest(challenge, solution)
    };
    
    if (validations.exactMatch) {
      return { valid: true, method: 'exact', confidence: 1.0 };
    }
    
    if (validations.regexMatch) {
      return { valid: true, method: 'pattern', confidence: 0.9 };
    }
    
    if (validations.functionalTest.passed) {
      return { 
        valid: true, 
        method: 'functional', 
        confidence: validations.functionalTest.confidence 
      };
    }
    
    return { 
      valid: false, 
      feedback: this.generateFeedback(challenge, solution),
      hints: this.getProgressiveHints(challenge, solution)
    };
  }
}

class AIIntegration {
  constructor(apiKey) {
    this.apiKey = apiKey;
    this.baseURL = 'https://generativelanguage.googleapis.com/v1beta/models/gemini-pro:generateContent';
    this.requestQueue = [];
    this.rateLimiter = new RateLimiter(60, 60000); // 60 requests per minute
    this.promptCache = new Map();
    this.fallbackHints = new Map();
  }
  
  async sendPrompt(prompt, context = {}) {
    try {
      // Check rate limiting
      if (!this.rateLimiter.allowRequest()) {
        return this.getFallbackResponse(prompt, context);
      }
      
      // Check cache for similar prompts
      const cacheKey = this.generateCacheKey(prompt, context);
      if (this.promptCache.has(cacheKey)) {
        return this.promptCache.get(cacheKey);
      }
      
      const requestBody = {
        contents: [{
          parts: [{
            text: this.buildContextualPrompt(prompt, context)
          }]
        }],
        generationConfig: {
          temperature: 0.3,
          topK: 40,
          topP: 0.95,
          maxOutputTokens: 1024,
        }
      };
      
      const response = await fetch(this.baseURL, {
        method: 'POST',
        headers: {
          'Content-Type': 'application/json',
          'x-goog-api-key': this.apiKey
        },
        body: JSON.stringify(requestBody)
      });
      
      if (!response.ok) {
        throw new Error(`API request failed: ${response.status}`);
      }
      
      const data = await response.json();
      const result = this.parseGeminiResponse(data);
      
      // Cache successful responses
      this.promptCache.set(cacheKey, result);
      
      return result;
      
    } catch (error) {
      console.error('AI request failed:', error);
      return this.getFallbackResponse(prompt, context);
    }
  }
  
  buildContextualPrompt(prompt, context) {
    const systemPrompt = `You are an AI coding assistant for Debug Quest, an educational game. 
    Your role is to provide helpful hints and guidance without giving direct solutions.
    
    Context:
    - Challenge Level: ${context.level || 'Unknown'}
    - Challenge Type: ${context.challengeType || 'General'}
    - Player's Current Code: ${context.currentCode || 'Not provided'}
    - Previous Attempts: ${context.attempts || 0}
    - Learning Path: ${context.learningPath || 'mixed'}
    
    Guidelines:
    - Provide hints that guide thinking, not complete solutions
    - Ask questions that lead to insights
    - Suggest debugging approaches and best practices
    - Encourage learning through discovery
    - Keep responses concise and actionable
    `;
    
    return `${systemPrompt}\n\nUser Question: ${prompt}`;
  }
  
  parseGeminiResponse(data) {
    try {
      const content = data.candidates?.[0]?.content?.parts?.[0]?.text;
      if (!content) throw new Error('Invalid response format');
      
      return {
        success: true,
        content: content.trim(),
        source: 'gemini',
        timestamp: Date.now()
      };
    } catch (error) {
      throw new Error('Failed to parse AI response');
    }
  }
  
  getFallbackResponse(prompt, context) {
    // Generate response from pre-written hints
    const fallbackHint = this.selectFallbackHint(prompt, context);
    return {
      success: true,
      content: fallbackHint,
      source: 'fallback',
      timestamp: Date.now()
    };
  }
  
  ratePromptQuality(prompt, outcome) {
    // Analyze prompt effectiveness for learning
    const qualityMetrics = {
      specificity: this.analyzeSpecificity(prompt),
      context: this.analyzeContext(prompt),
      clarity: this.analyzeClarity(prompt),
      outcome: outcome.success ? 1 : 0
    };
    
    const overallScore = Object.values(qualityMetrics).reduce((a, b) => a + b) / 4;
    
    return {
      score: overallScore,
      metrics: qualityMetrics,
      suggestions: this.generatePromptSuggestions(prompt, qualityMetrics)
    };
  }
}

class SacrificeSystem {
  constructor(gameState) {
    this.gameState = gameState;
    this.choiceHistory = [];
    this.consequences = new Map();
    this.tabSwitchTimer = null;
    this.isTabActive = true;
    this.setupTabMonitoring();
  }
  
  setupTabMonitoring() {
    // Non-invasive tab activity monitoring
    document.addEventListener('visibilitychange', () => {
      if (document.hidden) {
        this.handleTabSwitch();
      } else {
        this.handleTabReturn();
      }
    });
    
    window.addEventListener('blur', () => {
      this.handleTabSwitch();
    });
    
    window.addEventListener('focus', () => {
      this.handleTabReturn();
    });
  }
  
  handleTabSwitch() {
    if (!this.isTabActive) return; // Already tracking
    
    this.isTabActive = false;
    const currentTime = Date.now();
    
    // Start 30-second timer for reflection prompt
    this.tabSwitchTimer = setTimeout(() => {
      this.triggerReflectionPrompt('tab_switch');
    }, 30000);
  }
  
  handleTabReturn() {
    if (this.isTabActive) return; // Already active
    
    this.isTabActive = true;
    
    if (this.tabSwitchTimer) {
      clearTimeout(this.tabSwitchTimer);
      this.tabSwitchTimer = null;
    }
  }
  
  trackChoice(choiceType, context = {}) {
    const choice = {
      type: choiceType,
      timestamp: Date.now(),
      context: context,
      challengeId: context.challengeId,
      level: context.level
    };
    
    this.choiceHistory.push(choice);
    this.updateChoiceMetrics(choice);
    this.evaluateConsequences(choice);
    
    // Trigger reflection if pattern detected
    this.checkForPatterns();
  }
  
  updateChoiceMetrics(choice) {
    const currentChoices = this.gameState.state.choices;
    
    switch (choice.type) {
      case 'ai_assistance':
        currentChoices.aiAssistanceUsed++;
        break;
      case 'manual_solve':
        currentChoices.manualSolvesCompleted++;
        break;
      case 'hint_request':
        currentChoices.hintsRequested++;
        break;
      case 'tab_switch':
        currentChoices.tabSwitchDetected++;
        break;
    }
    
    this.gameState.updateState('choices', currentChoices);
  }
  
  evaluateConsequences(choice) {
    const recentChoices = this.getRecentChoices(5); // Last 5 choices
    const aiUsageRate = recentChoices.filter(c => c.type === 'ai_assistance').length / recentChoices.length;
    
    // Show consequences based on patterns
    if (aiUsageRate > 0.8) {
      this.scheduleConsequence('ai_dependency_warning', {
        message: "You've been relying heavily on AI assistance. The next challenge will test your independent problem-solving skills.",
        type: 'warning',
        delay: 2000
      });
    } else if (aiUsageRate < 0.2 && recentChoices.length >= 5) {
      this.scheduleConsequence('ai_underutilization', {
        message: "Great independent work! Remember that skilled developers also know when AI can boost productivity.",
        type: 'insight',
        delay: 1500
      });
    }
  }
  
  triggerReflectionPrompt(triggerType) {
    const prompts = {
      tab_switch: [
        "Noticed you switched tabs. Were you looking up solutions or taking a break?",
        "External research can be valuable. How will you apply what you learned?",
        "Taking breaks is important for learning. Ready to continue with fresh perspective?"
      ],
      ai_overuse: [
        "You've used AI assistance frequently. How confident do you feel about your understanding?",
        "AI can be helpful, but building independent skills is valuable too. Want to try the next challenge manually?",
        "Balancing AI help with personal growth is key. How do you want to approach the next problem?"
      ],
      manual_streak: [
        "Impressive problem-solving without AI! How are you feeling about the difficulty?",
        "You're building strong independent skills. Consider when AI might complement your abilities.",
        "Great persistence! Remember that efficient developers know when to use all available tools."
      ]
    };
    
    const promptOptions = prompts[triggerType] || prompts.tab_switch;
    const selectedPrompt = promptOptions[Math.floor(Math.random() * promptOptions.length)];
    
    this.showReflectionDialog(selectedPrompt, triggerType);
  }
  
  showReflectionDialog(prompt, type) {
    // Create non-blocking reflection interface
    const reflectionData = {
      prompt: prompt,
      type: type,
      timestamp: Date.now(),
      answered: false
    };
    
    // Update UI to show reflection prompt
    this.gameState.updateState('currentReflection', reflectionData);
  }
  
  getRecentChoices(count) {
    return this.choiceHistory.slice(-count);
  }
  
  calculateSacrificeScore() {
    const choices = this.gameState.state.choices;
    const totalChoices = choices.aiAssistanceUsed + choices.manualSolvesCompleted;
    
    if (totalChoices === 0) return 0;
    
    const aiDependencyRatio = choices.aiAssistanceUsed / totalChoices;
    const reflectionEngagement = choices.reflectionPrompts.filter(p => p.answered).length / Math.max(1, choices.reflectionPrompts.length);
    
    // Lower sacrifice score is better (more balanced approach)
    const balanceScore = Math.abs(aiDependencyRatio - 0.5); // Ideal is 50/50 balance
    const engagementBonus = reflectionEngagement * 0.1;
    
    return Math.max(0, balanceScore - engagementBonus);
  }
}

### *AI Prompt Quality System*
javascript
const PromptEvaluation = {
  criteria: {
    specificity: {
      weight: 0.3,
      evaluate: (prompt) => {
        const specificTerms = ['error', 'bug', 'issue', 'problem', 'not working', 'help with'];
        const hasContext = prompt.length > 20;
        const hasSpecificTerms = specificTerms.some(term => prompt.toLowerCase().includes(term));
        return hasContext && hasSpecificTerms ? 1 : 0.5;
      }
    },
    
    context: {
      weight: 0.4,
      evaluate: (prompt, challengeContext) => {
        const includesCode = prompt.includes(challengeContext.brokenCode?.substring(0, 50));
        const mentionsLanguage = prompt.toLowerCase().includes(challengeContext.language);
        const describesGoal = prompt.length > 50;
        return (includesCode ? 0.4 : 0) + (mentionsLanguage ? 0.3 : 0) + (describesGoal ? 0.3 : 0);
      }
    },
    
    clarity: {
      weight: 0.3,
      evaluate: (prompt) => {
        const wordCount = prompt.split(' ').length;
        const hasQuestion = prompt.includes('?');
        const isNotTooLong = wordCount < 100;
        const isNotTooShort = wordCount > 5;
        return (hasQuestion ? 0.3 : 0) + (isNotTooLong && isNotTooShort ? 0.7 : 0.3);
      }
    }
  },
  
  generateFeedback: (score, breakdown) => {
    const suggestions = [];
    
    if (breakdown.specificity < 0.7) {
      suggestions.push("Try to be more specific about the problem you're facing.");
    }
    
    if (breakdown.context < 0.7) {
      suggestions.push("Include relevant code snippets and mention the programming language.");
    }
    
    if (breakdown.clarity < 0.7) {
      suggestions.push("Ask a clear question and keep your prompt focused.");
    }
    
    return {
      overallScore: score,
      breakdown: breakdown,
      suggestions: suggestions,
      example: "Try: 'I have an HTML document that's not displaying correctly. The title appears to be cut off and there's a paragraph that's not showing. Can you help me identify which closing tags might be missing?'"
    };
  }
};
